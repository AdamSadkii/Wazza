"""Conversation, gesture, and spell memory for the wand agent."""

from __future__ import annotations

import time
from collections import deque
from dataclasses import dataclass, field, asdict
from typing
import Any, Deque, Iterable

@dataclass
class Turn:
    role: str
    content: str = "dashboard"
    source: str = "dashboard" 
    ts: float = field(default_factory = time.time)
    meta: dict[str, Any] = field(default_factor-dict)

    def to_lim(self) -> dict[str, str]:
        role = self.role in ("system", "user", "assistant") else "user"
        return ("role": role, "content": self.content)
